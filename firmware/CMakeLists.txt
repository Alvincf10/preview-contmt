cmake_minimum_required(VERSION 3.16)
project(ConMT LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CONMT_PROFILE HOST CACHE STRING "HOST only; production uses a verified Arduino board profile")
if(NOT CONMT_PROFILE STREQUAL "HOST")
  message(FATAL_ERROR "Use the Arduino build with a verified BoardProfile for production")
endif()
option(CONMT_REAL32 "Exercise binary32 arithmetic on host" OFF)
option(CONMT_SANITIZE "Enable Address/UndefinedBehavior sanitizers" OFF)
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS ConMT/src/core/*.cpp ConMT/src/storage/*.cpp ConMT/src/app/*.cpp ConMT/src/hmi/*.cpp)
add_library(conmt_core STATIC ${CORE_SOURCES})
target_include_directories(conmt_core PUBLIC ConMT/src)
target_compile_options(conmt_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CONMT_REAL32)
  target_compile_definitions(conmt_core PUBLIC CONMT_REAL32=1)
endif()
if(CONMT_SANITIZE)
  target_compile_options(conmt_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(conmt_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(conmt_host STATIC host/HostPorts.cpp)
target_link_libraries(conmt_host PUBLIC conmt_core)
target_include_directories(conmt_host PUBLIC host)
add_executable(conmt_tests tests/tests.cpp)
target_link_libraries(conmt_tests PRIVATE conmt_host)
add_executable(conmt_sim host/main.cpp)
target_link_libraries(conmt_sim PRIVATE conmt_host)
add_executable(conmt_oracle tests/oracle.cpp)
target_link_libraries(conmt_oracle PRIVATE conmt_core)
enable_testing()
add_test(NAME domain_storage_workflows COMMAND conmt_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME reference_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/compare_reference.py $<TARGET_FILE:conmt_oracle> $<BOOL:${CONMT_REAL32}>)
add_executable(conmt_workflow_tests tests/workflow.cpp)
target_link_libraries(conmt_workflow_tests PRIVATE conmt_host)
add_test(NAME workflows_faults_hmi COMMAND conmt_workflow_tests)
add_executable(conmt_audit_tests tests/audit.cpp)
target_link_libraries(conmt_audit_tests PRIVATE conmt_host)
add_test(NAME audit_regressions COMMAND conmt_audit_tests)
add_executable(conmt_reference_workflow tests/reference_workflow.cpp)
target_link_libraries(conmt_reference_workflow PRIVATE conmt_host)
target_compile_definitions(conmt_reference_workflow PRIVATE CONMT_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/excel.tsv")
add_test(NAME full_reference_workflow COMMAND conmt_reference_workflow)
add_executable(conmt_math_cases tests/math_cases.cpp)
target_link_libraries(conmt_math_cases PRIVATE conmt_core)
add_test(NAME mathematical_edge_cases COMMAND conmt_math_cases)
add_executable(conmt_battery_tests tests/battery.cpp)
target_link_libraries(conmt_battery_tests PRIVATE conmt_core)
add_test(NAME battery_calibration COMMAND conmt_battery_tests)
add_executable(conmt_index_tests tests/disk_index.cpp)
target_link_libraries(conmt_index_tests PRIVATE conmt_host)
add_test(NAME disk_index_recovery COMMAND conmt_index_tests)
add_test(NAME indexed_reference_workflow COMMAND conmt_reference_workflow --indexed)
add_executable(conmt_hmi_snapshot tests/hmi_snapshot.cpp)
target_link_libraries(conmt_hmi_snapshot PRIVATE conmt_host)
target_compile_definitions(conmt_hmi_snapshot PRIVATE CONMT_HMI_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/hmi-wire.txt")
add_test(NAME hmi_wire_snapshot COMMAND conmt_hmi_snapshot)
add_executable(conmt_real_codec_tests tests/real_codec.cpp)
target_link_libraries(conmt_real_codec_tests PRIVATE conmt_core)
add_test(NAME real_codec_precision COMMAND conmt_real_codec_tests)
add_executable(conmt_session_edges tests/session_edges.cpp)
target_link_libraries(conmt_session_edges PRIVATE conmt_host)
add_test(NAME session_clock_uart_edges COMMAND conmt_session_edges)
